If your suffix is in the list, the password has been breached.💻 Code Implementation (JavaScript / Node.js)Here is a clean, production-ready function to check a password using the official HIBP Pwned Passwords range API:javascriptasync function isPasswordLeaked(password) {
    // 1. Generate SHA-1 hash of the password
    const encoder = new TextEncoder();
    const data = encoder.encode(password);
    const hashBuffer = await crypto.subtle.digest('SHA-1', data);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    const sha1 = hashArray.map(b => b.toString(16).padStart(2, '0')).join('').toUpperCase();

    // 2. Split into 5-character prefix and 35-character suffix
    const prefix = sha1.slice(0, 5);
    const suffix = sha1.slice(5);

    // 3. Query the HIBP Range API (Requires a custom User-Agent)
    const url = `https://pwnedpasswords.com{prefix}`;
    const response = await fetch(url, {
        headers: { 'User-Agent': 'YourAppName-PasswordChecker' }
    });

    if (!response.ok) {
        throw new Error('Failed to fetch from HaveIBeenPwned API');
    }

    const text = await response.text();
    
    // 4. Parse the response lines (Format: "SUFFIX:COUNT")
    const lines = text.split('\n');
    for (const line of lines) {
        const [returnedSuffix, count] = line.trim().split(':');
        
        // 5. If suffix matches, the password is leaked
        if (returnedSuffix === suffix) {
            return { leaked: true, count: parseInt(count, 10) };
        }
    }

    return { leaked: false, count: 0 };
}
